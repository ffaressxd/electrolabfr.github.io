Smart Washing Machine

# Smart Washing Machine
# Automated System

class SmartWashingMachine:

    def __init__(self):
        self.weight = 0
        self.dirt_level = 0
        self.fabric = ""
        self.water = 0
        self.temperature = 20
        self.speed = 0
        self.energy = 0

    def detect_load(self):
        print("\n🔍 Detecting laundry...")
        
        if self.weight <= 3:
            self.water = 25
        elif self.weight <= 6:
            self.water = 40
        else:
            self.water = 55

        print(f"Load weight: {self.weight} kg")
        print(f"Water required: {self.water} L")

    def choose_program(self):
        print("\n🤖 AI is choosing the best program...")

        if self.fabric == "cotton":
            self.temperature = 40
            self.speed = 1000

        elif self.fabric == "wool":
            self.temperature = 30
            self.speed = 600

        elif self.fabric == "synthetic":
            self.temperature = 30
            self.speed = 800

        if self.dirt_level >= 8:
            self.temperature += 10

        print(f"Temperature: {self.temperature}°C")
        print(f"Spin speed: {self.speed} RPM")

    def washing(self):
        print("\n💧 Filling water...")
        print("🫧 Adding detergent...")
        print("🔄 Washing...")
        
        self.energy += self.temperature * 0.02

    def rinsing(self):
        print("\n🚿 Rinsing...")
        self.water += 20

    def spinning(self):
        print("\n🔄 Spinning...")
        self.energy += 0.15

    def finish(self):
        print("\n✅ Washing completed!")
        print(f"Water used: {self.water} L")
        print(f"Energy used: {self.energy:.2f} kWh")


machine = SmartWashingMachine()

machine.weight = float(input("Enter laundry weight (kg): "))
machine.dirt_level = int(input("Enter dirt level (1-10): "))
machine.fabric = input(
    "Enter fabric type (cotton / wool / synthetic): "
).lower()

machine.detect_load()
machine.choose_program()
machine.washing()
machine.rinsing()
machine.spinning()
machine.finish()
